bool emergenciaEnCurso(uint32_t* generacion) {
  const uint32_t ahora = (uint32_t)millis();
  portENTER_CRITICAL(&emergenciaMux);
  // La diferencia con signo sigue siendo válida al desbordar millis().
  if (emergenciaActiva && (int32_t)(ahora - emergenciaDeadline) >= 0) {
    emergenciaActiva = false;
    emergenciaGeneracion++;
  }
  bool activa = emergenciaActiva;
  if (generacion != nullptr) *generacion = emergenciaGeneracion;
  portEXIT_CRITICAL(&emergenciaMux);
  return activa;
}

bool aplicarEstadoEmergencia(JsonObjectConst estado, uint32_t demoraMs) {
  if (!estado["type"].is<const char*>() ||
      strcmp(estado["type"].as<const char*>(), "emergency") != 0 ||
      !estado["active"].is<bool>() || !estado["revision"].is<int64_t>() ||
      !estado["remaining_ms"].is<uint32_t>()) return false;

  const bool solicitada = estado["active"].as<bool>();
  const int64_t revision = estado["revision"].as<int64_t>();
  const char* id = estado["id"].is<const char*>() ? estado["id"].as<const char*>() : "";
  const size_t longitudId = strlen(id);
  if (revision < 0 || longitudId >= sizeof(emergenciaId) ||
      (solicitada && longitudId == 0)) return false;

  uint32_t restante = estado["remaining_ms"].as<uint32_t>();
  if (restante > 60000UL) restante = 60000UL;
  // Una respuesta HTTP demorada no extiende la vigencia indicada por el servidor.
  restante = restante > demoraMs ? restante - demoraMs : 0;
  const uint32_t ahora = (uint32_t)millis();

  portENTER_CRITICAL(&emergenciaMux);
  if (emergenciaActiva && (int32_t)(ahora - emergenciaDeadline) >= 0) {
    emergenciaActiva = false;
    emergenciaGeneracion++;
  }
  const bool mismoId = strcmp(emergenciaId, id) == 0;
  if (revision < emergenciaRevision || (revision == emergenciaRevision && !mismoId)) {
    portEXIT_CRITICAL(&emergenciaMux);
    return false;
  }

  const bool estabaActiva = emergenciaActiva;
  if (solicitada && restante > 0) {
    const uint32_t candidata = ahora + restante;
    if (!mismoId) {
      emergenciaActiva = true;
      emergenciaDeadline = candidata;
    } else if (emergenciaActiva && (int32_t)(candidata - emergenciaDeadline) < 0) {
      // Duplicados y consultas pueden acortar, pero nunca alargar el plazo.
      emergenciaDeadline = candidata;
    }
    // El mismo ID ya cerrado o vencido se conserva como marca de cierre.
  } else {
    emergenciaActiva = false;
  }
  if (estabaActiva != emergenciaActiva || (!mismoId && emergenciaActiva)) {
    emergenciaGeneracion++;
  }
  memcpy(emergenciaId, id, longitudId + 1);
  emergenciaRevision = revision;
  portEXIT_CRITICAL(&emergenciaMux);
  return true;
}

void recibirEstadoEmergenciaJSON(const String& payload, uint32_t demoraMs) {
  StaticJsonDocument<768> doc;
  if (deserializeJson(doc, payload)) return;
  JsonObjectConst estado = doc.as<JsonObjectConst>();
  if (doc["emergencia"].is<JsonObjectConst>()) {
    if (!doc["success"].is<bool>() || !doc["success"].as<bool>()) return;
    estado = doc["emergencia"].as<JsonObjectConst>();
  }
  aplicarEstadoEmergencia(estado, demoraMs);
}

// Tono del aviso con un búfer propio.
void reproducirTonoAvisoAPI() {
  int16_t avisoStereoFrame[RX_DAC_SAMPLES * 2];

  for (int i = 0; i < RX_DAC_SAMPLES; i++) {
    int16_t sample = ((i / 12) % 2 == 0) ? 14000 : -14000;
    avisoStereoFrame[i * 2] = sample;
    avisoStereoFrame[i * 2 + 1] = sample;
  }

  for (int b = 0; b < 8; b++) {
    size_t written = 0;
    i2s_write(
      I2S_PORT,
      avisoStereoFrame,
      sizeof(avisoStereoFrame),
      &written,
      pdMS_TO_TICKS(6)
    );
  }
}

String upperTrim(String s) {
  s.trim();
  s.replace("\r", "");
  s.replace("\n", "");
  s.trim();
  s.toUpperCase();
  return s;
}

String jsonEscape(const char* text) {
  String out;
  if (!text) return out;
  while (*text) {
    char c = *text++;
    if (c == '"') out += "\\\"";
    else if (c == '\\') out += "\\\\";
    else if (c == '\n') out += "\\n";
    else if (c == '\r') out += "\\r";
    else out += c;
  }
  return out;
}

String jsonStringValue(const String &json, const String &key) {
  String needle = String("\"") + key + "\"";
  int p = json.indexOf(needle);
  if (p < 0) return "";
  p = json.indexOf(':', p + needle.length());
  if (p < 0) return "";
  p++;
  while (p < (int)json.length() && (json[p] == ' ' || json[p] == '\t')) p++;
  if (p >= (int)json.length()) return "";
  if (json[p] == 'n') return "";
  if (json[p] != '\"') return "";
  p++;
  String out;
  bool esc = false;
  for (; p < (int)json.length(); p++) {
    char c = json[p];
    if (esc) { out += c; esc = false; }
    else if (c == '\\') { esc = true; }
    else if (c == '\"') { break; }
    else { out += c; }
  }
  return out;
}

size_t alignRxFrame(size_t n) {
  return ((n + RX_FRAME_BYTES - 1) / RX_FRAME_BYTES) * RX_FRAME_BYTES;
}

size_t ringAvailableUnsafe() {
  if (rxHead >= rxTail) return rxHead - rxTail;
  return RX_RING_SIZE - rxTail + rxHead;
}

size_t ringAvailable() {
  portENTER_CRITICAL(&ringMux);
  size_t n = ringAvailableUnsafe();
  portEXIT_CRITICAL(&ringMux);
  return n;
}

size_t ringFreeUnsafe() {
  return RX_RING_SIZE - 1 - ringAvailableUnsafe();
}

void ringDropOldestAlignedUnsafe(size_t len) {
  size_t available = ringAvailableUnsafe();
  if (available == 0) return;
  len = alignRxFrame(len);
  if (len > available) len = available - (available % RX_FRAME_BYTES);
  if (len == 0) return;
  rxTail = (rxTail + len) % RX_RING_SIZE;
  droppedRxBytes += len;
}

bool ringPushKeepNewestAligned(const uint8_t* data, size_t len) {
  len -= (len % 2);
  if (len == 0) return false;
  if (len >= RX_RING_SIZE) {
    size_t keep = RX_RING_SIZE - RX_FRAME_BYTES;
    keep -= keep % RX_FRAME_BYTES;
    data += (len - keep);
    len = keep;
  }
  portENTER_CRITICAL(&ringMux);
  size_t freeBytes = ringFreeUnsafe();
  if (len > freeBytes) ringDropOldestAlignedUnsafe(len - freeBytes);
  for (size_t i = 0; i < len; i++) {
    rxRing[rxHead] = data[i];
    rxHead = (rxHead + 1) % RX_RING_SIZE;
  }
  size_t available = ringAvailableUnsafe();
  if (available > RX_TRIM_BUFFER_BYTES) {
    size_t drop = available - RX_TARGET_BUFFER_BYTES;
    drop -= (drop % RX_FRAME_BYTES);
    ringDropOldestAlignedUnsafe(drop);
  }
  portEXIT_CRITICAL(&ringMux);
  return true;
}

size_t ringPop(uint8_t* out, size_t maxLen) {
  portENTER_CRITICAL(&ringMux);
  size_t available = ringAvailableUnsafe();
  size_t n = (available < maxLen) ? available : maxLen;
  n -= (n % 2);
  for (size_t i = 0; i < n; i++) {
    out[i] = rxRing[rxTail];
    rxTail = (rxTail + 1) % RX_RING_SIZE;
  }
  portEXIT_CRITICAL(&ringMux);
  return n;
}

void ringClear() {

  portEXIT_CRITICAL(&ringMux);
}

void trimRxLatencyIfNeeded() {
  portENTER_CRITICAL(&ringMux);
  size_t available = ringAvailableUnsafe();
  if (available > RX_TRIM_BUFFER_BYTES) {
    size_t drop = available - RX_TARGET_BUFFER_BYTES;
    drop -= (drop % RX_FRAME_BYTES);
    ringDropOldestAlignedUnsafe(drop);
  }
  portEXIT_CRITICAL(&ringMux);
}

bool httpRequest(const String &url, const String &method, const String &body, String &response, int &statusCode) {
  if (!isNetworkConnected()) return false;
  WiFiClientSecure client;
  client.setInsecure();
  HTTPClient http;
  http.setTimeout(HTTP_TIMEOUT_MS);
  if (!http.begin(client, url)) { statusCode = -1; return false; }
  if (method == "POST") {
    http.addHeader("Content-Type", "application/json");
    statusCode = http.POST(body);
  } else {
    statusCode = http.GET();
  }
  response = http.getString();
  http.end();
  return statusCode >= 200 && statusCode < 300;
}

bool postJson(const String &path, const String &body, String &response, int &statusCode) {
  return httpRequest(String(BACKEND_BASE_URL) + path, "POST", body, response, statusCode);
}

bool getPath(const String &path, String &response, int &statusCode) {
  return httpRequest(String(BACKEND_BASE_URL) + path, "GET", "", response, statusCode);
}

bool parseSetWifiCommand(const String &line, String &ssid, String &pass) {
  const String prefix = "SET_WIFI:";
  if (!line.startsWith(prefix)) return false;
  String payload = line.substring(prefix.length());
  int sep = payload.indexOf(':');
  if (sep < 0) return true;
  ssid = payload.substring(0, sep);
  pass = payload.substring(sep + 1);
  ssid.trim(); pass.trim();
  return true;
}

bool parseWifiJsonCommand(const String &line, String &ssid, String &pass) {
  const String prefix = "WIFI_JSON:";
  if (!line.startsWith(prefix)) return false;
  String json = line.substring(prefix.length());
  ssid = jsonStringValue(json, "ssid");
  pass = jsonStringValue(json, "pass");
  ssid.trim(); pass.trim();
  return true;
}

bool handleWifiCommand(String line) {
  line.trim();
  if (line.length() == 0) return false;
  String upper = upperTrim(line);

  if (upper == "WIFI_HELP") { printWiFiHelp(); return true; }
  if (upper == "WIFI_STATUS") { printWiFiStatus(); return true; }
  if (upper == "WIFI_RECONNECT") { requestWiFiReconnect(); return true; }
  if (upper == "WIFI_RESET") { clearWiFiConfig(); requestWiFiReconnect(); return true; }

  String ssid, pass;
  if (parseSetWifiCommand(line, ssid, pass) || parseWifiJsonCommand(line, ssid, pass)) {
    if (ssid.length() == 0) return true;
    if (saveWiFiConfig(ssid, pass)) requestWiFiReconnect();
    return true;
  }
  return false;
}

  if (!wifiReconnectRequested) return;
  if (callWanted || incomingRinging) return;
  wifiReconnectRequested = false;
  Serial.println("[WIFI] Reconectando...");
  wsRxConnected = false;
  wsTxCloseRequested = true;
  WiFi.disconnect(true);
  delay(250);
  connectWiFi();
}

void connectWiFi() {
  if (WiFi.status() == WL_CONNECTED) return;
  
  wifiConnecting = true;
  
  // Limpiar estado previo para evitar errores de config superpuesta
  WiFi.disconnect(true, true);
  delay(500);

  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);

  // Si el usuario configuró una red personalizada por preferencias, intentarla primero
  if (wifiSsid.length() > 0) {
    Serial.print("[WIFI] Intentando red guardada: "); Serial.println(wifiSsid);
    WiFi.begin(wifiSsid.c_str(), wifiPass.c_str());
    uint32_t start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < wifi_timeout) {
      delay(250); Serial.print(".");
    }
    Serial.println();
    if (WiFi.status() == WL_CONNECTED) {
      Serial.print("WiFi conectado. IP: "); Serial.println(WiFi.localIP());
      wifiConnecting = false;
      return;
    }
    Serial.println("[WIFI] Falló red guardada.");
    
    WiFi.disconnect(true, true);
    delay(500);
  }

  // Recorrido de la lista de respaldo (Failover automático)
  for (int i = 0; i < total_networks; i++) {
    Serial.printf("[WIFI] Intentando red de lista [%d]: %s\n", i, ssid_list[i]);
    WiFi.begin(ssid_list[i], password_list[i]);

    unsigned long start_time = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start_time < wifi_timeout) {
      delay(250);
      Serial.print(".");
    }
    Serial.println();

    if (WiFi.status() == WL_CONNECTED) {
      Serial.println("[WIFI] ¡Conectado exitosamente a la red!");
      Serial.print("Dirección IP: ");
      Serial.println(WiFi.localIP());
      wifiConnecting = false;
      return;
    } else {
      Serial.printf("[WIFI] Falló la conexión con %s\n", ssid_list[i]);
      WiFi.disconnect(true, true);
      delay(500);
    }
  }

  Serial.println("[WIFI] Error crítico: No se pudo conectar a ninguna red Wi-Fi.");
  wifiConnecting = false;
}

void onRxMessageCallback(WebsocketsMessage message) {
  // Los avisos llegan como texto WebSocket.
  if (message.isText()) {
    String texto = message.data();

    if (texto.startsWith(EMERGENCIA_APP_PREFIX)) {
      recibirEstadoEmergenciaJSON(texto.substring(sizeof(EMERGENCIA_APP_PREFIX) - 1), 0);
    } else if (texto.startsWith(AVISO_APP_PREFIX)) {
      portENTER_CRITICAL(&avisoMux);
      avisoLedPendiente = true;
      avisoTonoPendiente = true;
      portEXIT_CRITICAL(&avisoMux);

      Serial.print("[AVISO APP] ");
      Serial.println(
        texto.substring(sizeof(AVISO_APP_PREFIX) - 1)
      );
    }

    // Ningún mensaje de texto debe entrar al búfer de audio.
    return;
  }

  // Mantener el procesamiento original del audio durante llamadas.
  if (!message.isBinary() || !callWanted) return;

  std::string raw = message.rawData();
  if (raw.empty()) return;
  
  if (callWanted && !callActive) {
    callActive = true;
    Serial.println("[CALL] LLAMADA ACTIVA CONECTADA");
  }
  wsRxBytes += raw.size();
  if (emergenciaEnCurso(nullptr)) return; // La voz sigue llegando, sin acumularla durante la sirena.
  ringPushKeepNewestAligned((const uint8_t*)raw.data(), raw.size());
}

void solicitarAperturaPuerta(uint8_t puerta) {
  if (puerta != 1 && puerta != 2) return;

  if (!isNetworkConnected() || colaPuertas == nullptr) {
    Serial.println("[PUERTAS] No se puede enviar: sin WiFi o sin cola.");
    return;
  }

  SolicitudPuerta solicitud = {puerta, millis()};

  if (xQueueSend(colaPuertas, &solicitud, 0) != pdPASS) {
    Serial.println("[PUERTAS] Cola llena. Solicitud descartada.");
    return;
  }

  Serial.printf("[PUERTAS] Solicitud en cola: puerta %u\n", puerta);
}

void procesarSolicitudesPuertas() {
  if (!wsTxConnected || colaPuertas == nullptr) return;

  SolicitudPuerta solicitud;
  if (xQueueReceive(colaPuertas, &solicitud, 0) != pdPASS) return;

  if (millis() - solicitud.creadaEn > PUERTA_SOLICITUD_MAX_MS) {
    Serial.println("[PUERTAS] Solicitud vencida. Vuelve a pulsar.");
    return;
  }

  StaticJsonDocument<192> doc;
  doc["device_id"] = ESP32_DEVICE_ID;
  doc["accion"] = "abrir";
  doc["puerta"] = solicitud.puerta;

  String payload;
  serializeJson(doc, payload);

  if (wsTxClient.send(payload)) {
    Serial.print("[PUERTAS] Solicitud enviada: ");
    Serial.println(payload);
  } else {
    Serial.println("[PUERTAS] No se pudo confirmar el envio.");
  }
}

void beginIncomingCallFromBackend(const String &response) {
  if (callWanted || callActive) return;

  String callId = jsonStringValue(response, "call_id");
  String callerRut = jsonStringValue(response, "caller_rut");
  String callerDpto = jsonStringValue(response, "caller_dpto");

  if (callId.length() == 0) return;
  if (currentCallId == callId && incomingRinging) return;

  currentCallId = callId;
  currentPeerRut = callerRut;
  currentPeerDpto = callerDpto.length() ? callerDpto : callerRut;
  incomingRinging = true;
  callWanted = false;
  callActive = false;
  callStartedAt = 0;
  ringClear();

  Serial.print("[CALL] *** LLAMADA ENTRANTE call_id=");
  Serial.print(currentCallId);
  Serial.print(" caller=");
  Serial.println(currentPeerDpto);
}

void acceptIncomingCallOnBackend() {
  if (!incomingRinging || currentCallId.length() == 0) return;

  String body = String("{") +
    "\"call_id\":\"" + currentCallId + "\"," +
    "\"device_id\":\"" + ESP32_DEVICE_ID + "\"" +
    "}";

  String response;
  int code = 0;
  bool ok = postJson("/api/esp32/answer-call", body, response, code);

  Serial.print("[CALL] POST /api/esp32/answer-call code=");
  Serial.print(code); Serial.print(" ok="); Serial.println(ok ? "true" : "false");

  String callId = jsonStringValue(response, "call_id");
  if (callId.length() > 0) currentCallId = callId;

  if (!ok) { stopLocalCall(true); return; }

  incomingRinging = false;
  setAudioCallStartFlags();
  callActive = true;
  Serial.println("[CALL] INICIAR LLAMADA");
}

void rejectIncomingCallOnBackend() {
  if (!incomingRinging) return;
  String body = String("{") +
    "\"call_id\":\"" + currentCallId + "\"," +
    "\"device_id\":\"" + ESP32_DEVICE_ID + "\"" +
    "}";
  String response; int code = 0;
  postJson("/api/esp32/reject-call", body, response, code);
  stopLocalCall(true);
}

void handleRemoteCallEnded(const String &response) {
  String endedCallId = jsonStringValue(response, "call_id");
  if (currentCallId.length() && endedCallId.length() && endedCallId != currentCallId) return;
  Serial.println("[CALL] Backend indica fin/cancelacion remota");
  stopLocalCall(true);
}

void pollBackendCallState() {
  if (!isNetworkConnected()) return;
  
  if (callWanted && !wsRxConnected && !wsTxConnected) {
     return; 
  }

  // Una respuesta lenta no debe alterar una llamada que ya cambio.
  const String polledCallId = currentCallId;
  String path = String("/api/esp32/poll?device_id=") + ESP32_DEVICE_ID;
  if (polledCallId.length() > 0) path += "&call_id=" + polledCallId;

  String response; int code = 0;
  bool ok = getPath(path, response, code);
  if (!ok || currentCallId != polledCallId) return;

  String action = jsonStringValue(response, "action");

  if (action == "idle") {
    if (currentCallId.length() > 0 && (callWanted || callActive || incomingRinging)) {
      Serial.println("[CALL] Backend sin llamada vigente: liberando estado local.");
      stopLocalCall(true);
    }
    return;
  }

  if (action == "incoming_call") { beginIncomingCallFromBackend(response); return; }
  if (action == "active") {
    if (callWanted && !callActive) {
      callActive = true;
      callStartedAt = millis();
      Serial.println("[CALL] INICIAR LLAMADA DESDE BACKEND");
    }
    return;
  }
  if (action == "end_call") { handleRemoteCallEnded(response); return; }
}

void writeIncomingRingToneFrame() {
  static uint32_t sampleCounter = 0;
  bool toneOn = (millis() % 3000UL) < 1000UL;
  for (int i = 0; i < RX_DAC_SAMPLES; i++) {
    int16_t sample = 0;
    // Utiliza la constante RING_TONE_AMPLITUDE definida arriba
    if (toneOn) sample = ((sampleCounter / 18) % 2 == 0) ? RING_TONE_AMPLITUDE : -RING_TONE_AMPLITUDE; 
    sampleCounter++;
    dacStereoFrame[i * 2 + 0] = sample;
    dacStereoFrame[i * 2 + 1] = sample;
  }
  size_t written = 0;
  esp_err_t err = i2s_write(I2S_PORT, dacStereoFrame, sizeof(dacStereoFrame), &written, pdMS_TO_TICKS(6));
  if (err == ESP_OK) dacWrittenBytes += written;
}

// Timbre exclusivo de emergencia: 500 ms de tono y 250 ms de pausa.
void writeEmergencyRingToneFrame() {
  static uint32_t sampleCounter = 0;
  bool toneOn = (millis() % 750UL) < 500UL;
  for (int i = 0; i < RX_DAC_SAMPLES; i++) {
    int16_t sample = 0;
    if (toneOn) sample = ((sampleCounter / 18) % 2 == 0) ? RING_TONE_AMPLITUDE : -RING_TONE_AMPLITUDE;
    sampleCounter++;
    dacStereoFrame[i * 2 + 0] = sample;
    dacStereoFrame[i * 2 + 1] = sample;
  }
  size_t written = 0;
  esp_err_t err = i2s_write(I2S_PORT, dacStereoFrame, sizeof(dacStereoFrame), &written, pdMS_TO_TICKS(6));
  if (err == ESP_OK) dacWrittenBytes += written;
}

// === TONO SUAVE Y AGRADABLE PARA BOTÓN LLAMADA (GPIO 1) ===
void playCallBeep() {
  for (int b = 0; b < 12; b++) {
    for (int i = 0; i < RX_DAC_SAMPLES; i++) {
      int16_t sample = ((i / 32) % 2 == 0) ? 12000 : -12000;
      dacStereoFrame[i * 2 + 0] = sample;
      dacStereoFrame[i * 2 + 1] = sample;
    }
    size_t written = 0;
    i2s_write(I2S_PORT, dacStereoFrame, sizeof(dacStereoFrame), &written, pdMS_TO_TICKS(6));
  }
}

// === TONO DISTINTO Y SUAVE PARA BOTÓN MENSAJE / ALERTA (GPIO 2) ===
void playMessageBeep() {
  for (int b = 0; b < 8; b++) {
    for (int i = 0; i < RX_DAC_SAMPLES; i++) {
      int16_t sample = ((i / 12) % 2 == 0) ? 14000 : -14000;
      dacStereoFrame[i * 2 + 0] = sample;
      dacStereoFrame[i * 2 + 1] = sample;
    }
    size_t written = 0;
    i2s_write(I2S_PORT, dacStereoFrame, sizeof(dacStereoFrame), &written, pdMS_TO_TICKS(6));
  }
}

void audioPlaybackTask(void* parameter) {
  (void)parameter;
  bool primed = false;
  bool emergenciaSonando = false;
  uint32_t generacionReproducida = 0;
  const TickType_t framePeriod = pdMS_TO_TICKS(20);
  TickType_t lastWake = xTaskGetTickCount();

  for (;;) {
    uint32_t generacion = 0;
    const bool reproducirEmergencia = emergenciaEnCurso(&generacion);
    if (reproducirEmergencia != emergenciaSonando || generacion != generacionReproducida) {
      ringClear();
      i2s_zero_dma_buffer(I2S_PORT);
      primed = false;
      emergenciaSonando = reproducirEmergencia;
      generacionReproducida = generacion;
      lastWake = xTaskGetTickCount();
    }
    if (reproducirEmergencia) {
      writeEmergencyRingToneFrame();
      vTaskDelayUntil(&lastWake, framePeriod);
      continue;
    }

    bool emitirTonoAviso = false;

    if (!callWanted && !callActive && !incomingRinging) {
      portENTER_CRITICAL(&avisoMux);
      emitirTonoAviso = avisoTonoPendiente;
      avisoTonoPendiente = false;
      portEXIT_CRITICAL(&avisoMux);
    }

    if (emitirTonoAviso) {
      reproducirTonoAvisoAPI();
      lastWake = xTaskGetTickCount();
    }

    if (incomingRinging && !callWanted) {
      primed = false;
      writeIncomingRingToneFrame();
      vTaskDelayUntil(&lastWake, framePeriod);
      continue;
    }

    // === TIMBRE DE LLAMADA SALIENTE (RING RING LOCAL) ===
    if (callWanted && !callActive) {
      primed = false;
      writeIncomingRingToneFrame(); 
      vTaskDelayUntil(&lastWake, framePeriod);
      continue;
    }

    if (!callWanted) {
      primed = false;
      lastWake = xTaskGetTickCount();
      vTaskDelay(pdMS_TO_TICKS(10));
      continue;
    }
    trimRxLatencyIfNeeded();
    size_t available = ringAvailable();
    if (!primed) {
      if (available < RX_TARGET_BUFFER_BYTES) {
        memset(dacStereoFrame, 0, sizeof(dacStereoFrame));
        size_t written = 0;
        i2s_write(I2S_PORT, dacStereoFrame, sizeof(dacStereoFrame), &written, pdMS_TO_TICKS(4));
        if (written > 0) dacWrittenBytes += written;
        vTaskDelayUntil(&lastWake, framePeriod);
        continue;
      }
      primed = true;
    }
    bool haveAudio = false;
    if (ringAvailable() >= RX_FRAME_BYTES) {
      size_t n = ringPop(rxFrameMonoBytes, RX_FRAME_BYTES);
      haveAudio = (n == RX_FRAME_BYTES);
    }
    if (!haveAudio) {
      rxUnderruns++;
      memset(dacStereoFrame, 0, sizeof(dacStereoFrame));
    } else {
      int16_t* mono = (int16_t*)rxFrameMonoBytes;
      int out = 0;
      float gain = SPEAKER_GAIN;

      for (int i = 0; i < RX_FRAME_SAMPLES; i++) {
        int32_t amplified = (int32_t)(mono[i] * gain);
        
        if (amplified > 32767) amplified = 32767;
        if (amplified < -32768) amplified = -32768;
        
        int16_t s = (int16_t)amplified;

        dacStereoFrame[out++] = s;
        dacStereoFrame[out++] = s;
        dacStereoFrame[out++] = s;
        dacStereoFrame[out++] = s;
      }
    }
    size_t written = 0;
    i2s_write(I2S_PORT, dacStereoFrame, sizeof(dacStereoFrame), &written, pdMS_TO_TICKS(6));
    if (written > 0) dacWrittenBytes += written;
    vTaskDelayUntil(&lastWake, framePeriod);
  }
}

int16_t amplifyAudioForApp(int32_t sample) {
  sample = (sample * APP_TX_AUDIO_VOLUME_Q8) >> 8;
  if (sample > 32767) return 32767;
  if (sample < -32768) return -32768;
  return (int16_t)sample;
}

void sendMicToBrowser() {
  if (!callActive || !wsTxConnected || wsTxCloseRequested) return;
  size_t bytesIn = 0;
  esp_err_t result = i2s_read(I2S_PORT, (void*)i2sTxStereoFrame, sizeof(i2sTxStereoFrame), &bytesIn, pdMS_TO_TICKS(80));
  if (result != ESP_OK) { txReadErrors++; return; }
  if (bytesIn < sizeof(i2sTxStereoFrame)) { txReadShorts++; return; }

  for (int i = 0; i < TX_NET_SAMPLES; i++) {
    int32_t a = i2sTxStereoFrame[i * 2 * 2 + 0];
    int32_t b = i2sTxStereoFrame[(i * 2 + 1) * 2 + 0];
    int32_t mixed = (a + b) / 2;
    micTxMonoFrame[i] = amplifyAudioForApp(mixed);
  }
  if (!callActive || !wsTxConnected || wsTxCloseRequested) return;

  bool ok = wsTxClient.sendBinary((const char*)micTxMonoFrame, sizeof(micTxMonoFrame));
  if (ok) { 
    wsTxBytes += sizeof(micTxMonoFrame); 
  } else { 
    txSendFails++; 
    if (!wsTxClient.available()) {
      wsTxConnected = false;
    }
  }
}

void audioMicTxTask(void* parameter) {
  (void)parameter;

  for (;;) {
    if (!isNetworkConnected() || wsTxCloseRequested) {
      if (wsTxConnected) { wsTxClient.close(); }
      wsTxConnected = false;
      wsTxCloseRequested = false;
      if (!isNetworkConnected() && colaPuertas != nullptr) { xQueueReset(colaPuertas); }
      vTaskDelay(pdMS_TO_TICKS(20));
      continue;
    }
    if (!wsTxConnected && millis() - lastTxReconnectAttempt > WS_RECONNECT_MS) {
      lastTxReconnectAttempt = millis();
      connectWsTx();
    }
    if (wsTxConnected) {
      wsTxClient.poll();
      procesarSolicitudesPuertas();
      if (callActive) { sendMicToBrowser(); }
      wsTxClient.poll();
    }
    vTaskDelay(pdMS_TO_TICKS(2));
  }
}

void audioWsRxTask(void* parameter) {
  (void)parameter;
  for (;;) {
    if (!isNetworkConnected() || !callWanted && !RX_AVISOS_EN_REPOSO) {
      if (wsRxConnected) { wsRxClient.close(); wsRxConnected = false; }
      vTaskDelay(pdMS_TO_TICKS(20));
      continue;
    }
    if (!wsRxConnected && millis() - lastRxReconnectAttempt > WS_RECONNECT_MS) {
      lastRxReconnectAttempt = millis();
      connectWsRx();
    }
    if (wsRxConnected) wsRxClient.poll();
    vTaskDelay(pdMS_TO_TICKS(2));
  }
}

void backendPollTask(void* parameter) {
  (void)parameter;
  for (;;) {
    if (isNetworkConnected()) {
      unsigned long now = millis();
      if (now - lastEmergenciaPoll >= EMERGENCIA_POLL_MS) {
        lastEmergenciaPoll = now;
        pollBackendEmergencyState();
        now = millis();
      }
      if (now - lastEsp32Poll >= ESP32_POLL_MS) {
        lastEsp32Poll = now;
        pollBackendCallState();
      }
    }
    vTaskDelay(pdMS_TO_TICKS(50));
  }
}

bool apiCrearRecadoDesdeAlerta() {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("[RECADO] No hay conexión WiFi.");
    return false;
  }

  HTTPClient http;
  String url = "http://" + String(BACKEND_HOST) +
               ":" + String(BACKEND_PORT) + "/api/recados";

  http.setTimeout(HTTP_TIMEOUT_MS);

  if (!http.begin(url)) {
    Serial.println("[RECADO] No se pudo iniciar la petición HTTP.");
    return false;
  }

  http.addHeader("Content-Type", "application/json");

  StaticJsonDocument<512> doc;
  doc["rut_emisor"] = DEVICE_RUT;
  doc["origen"] = "citofono";
  doc["device_id"] = ESP32_DEVICE_ID;
  doc["titulo"] = "Solicitud de recado del citofono";
  doc["descripcion"] =
    "Tienes un recado pendiente del usuario " + String(DEVICE_RUT) + ", por favor llamarlo.";

  String body;
  serializeJson(doc, body);

  Serial.println("---------- ENVIO DE RECADO ----------");
  Serial.println("URL enviada: " + url);
  Serial.println("Datos enviados: " + body);

  int httpCode = http.POST(body);
  Serial.printf("[RECADO] Codigo HTTP: %d\n", httpCode);

  if (httpCode > 0) {
    Serial.println("[RECADO] Respuesta: " + http.getString());
  } else {
    Serial.println(
      "[RECADO] Error de conexion: " +
      HTTPClient::errorToString(httpCode)
    );
  }

  bool enviado = (httpCode == HTTP_CODE_OK || httpCode == 201);
  http.end();

  if (enviado) {
    Serial.println("[RECADO] Recado pendiente creado correctamente.");
  } else {
    Serial.println("[RECADO] No se confirmó la creación del recado.");
  }

  return enviado;
}

// Lógica de acción según el estado actual del citófono al pulsar el botón
void handleButtonPress() {
  Serial.println("[BOTON] Presionado GPIO 1");
  ledEnclavado = false; // Desenclava el LED al presionar el botón principal
  if (incomingRinging) {
    answerIncomingFromButton();
  } else if (callWanted || callActive) {
    endCallFromButton();
  } else {
    startCallFromButton();
  }
}

void processButtonOrCmdLine(String line) {
  line.trim();
  if (line.length() == 0) return;

  if (handleWifiCommand(line)) return;

  String cmd = upperTrim(line);

  if (cmd == "LLAMAR") { startCallFromButton(); return; }
  if (cmd == "COLGAR" || cmd == "FINALIZAR_LLAMADA") { endCallFromButton(); return; }
  if (cmd == "CONTESTAR") { answerIncomingFromButton(); return; }
  if (cmd == "ALERTA") { apiEnviarAlerta(); return; }

  // Pruebas directas
  if (cmd == "TEST_INCOMING") { simulateTestIncomingCall(); return; }
  if (cmd == "TEST_ACCEPT") { simulateTestAcceptCall(); return; }
  if (cmd == "TEST_REJECT") { simulateTestRejectCall(); return; }
  if (cmd == "TEST_END") { simulateTestEndCall(); return; }
}

void checkButtonPin() {
  static bool lastState = HIGH;
  static unsigned long lastDebounceTime = 0;
  static unsigned long pressStartTime = 0;
  static bool holdTriggered = false;
  
  bool reading = digitalRead(BUTTON_PIN);

  if (reading != lastState) {
    lastDebounceTime = millis();
  }

  if ((millis() - lastDebounceTime) > 50) {
    static bool buttonState = HIGH;
    if (reading != buttonState) {
      buttonState = reading;
      if (buttonState == LOW) {
        pressStartTime = millis();
        holdTriggered = false;
      } else {
        if (!holdTriggered) {
          handleButtonPress();
        }
      }
    } else if (buttonState == LOW && !holdTriggered) {
      if (incomingRinging && (millis() - pressStartTime >= 3000)) {
        Serial.println("[BOTON] GPIO 1 mantenido 3s -> Rechazando llamada entrante");
        rejectIncomingCallOnBackend();
        holdTriggered = true;
      }
    }
  }
  lastState = reading;
}

void checkButtonPin2() {
  static bool lastState2 = HIGH;
  static unsigned long lastDebounceTime2 = 0;
  
  bool reading2 = digitalRead(BUTTON_PIN_2);

  if (reading2 != lastState2) {
    lastDebounceTime2 = millis();
  }

  if ((millis() - lastDebounceTime2) > 50) {
    static bool buttonState2 = HIGH;
    if (reading2 != buttonState2) {
      buttonState2 = reading2;
      if (buttonState2 == LOW) {
        Serial.println("[BOTON 2] Presionado GPIO 2");
        if (ledEnclavado) {
          ledEnclavado = false; // Si ya estaba enclavado, se apaga al presionarlo de nuevo
          Serial.println("[LED] Desenclavado manualmente con el botón de aviso.");
        } else {
          apiEnviarAlerta();           // Envía el mensaje de chat y enclava el LED
          apiCrearRecadoDesdeAlerta(); // Crea el recado.
        }
      }
    }
  }
  lastState2 = reading2;
}

// === COMPROBACIÓN DE BOTONES DE LAS PUERTAS ===
void checkDoorButtons() {
  static bool lastDoor1State = HIGH;
  static unsigned long lastDebounceDoor1 = 0;
  bool readingDoor1 = digitalRead(DOOR_BUTTON_1);

  if (readingDoor1 != lastDoor1State) {
    lastDebounceDoor1 = millis();
  }
  if ((millis() - lastDebounceDoor1) > 50) {
    static bool door1State = HIGH;
    if (readingDoor1 != door1State) {
      door1State = readingDoor1;
      if (door1State == LOW) {
        Serial.println("[PUERTA 1] Presionado GPIO 21 - Apertura Puerta 1");
        solicitarAperturaPuerta(1);
      }
    }
  }
  lastDoor1State = readingDoor1;

  static bool lastDoor2State = HIGH;
  static unsigned long lastDebounceDoor2 = 0;
  bool readingDoor2 = digitalRead(DOOR_BUTTON_2);

  if (readingDoor2 != lastDoor2State) {
    lastDebounceDoor2 = millis();
  }
  if ((millis() - lastDebounceDoor2) > 50) {
    static bool door2State = HIGH;
    if (readingDoor2 != door2State) {
      door2State = readingDoor2;
      if (door2State == LOW) {
        Serial.println("[PUERTA 2] Presionado GPIO 20 - Apertura Puerta 2");
        solicitarAperturaPuerta(2);
      }
    }
  }
  lastDoor2State = readingDoor2;
}

// === GESTIÓN DEL LED (PARPADEO AL LLAMAR, BUSCANDO RED O FIJO SI ESTÁ ENCLAVADO) ===
void handleLedControl() {
  // 1. Prioridad: Parpadeo durante llamadas entrantes, salientes o activas
  if (callWanted || incomingRinging) {
    if (millis() - lastLedBlinkTime >= 500) {
      lastLedBlinkTime = millis();
      ledBlinkState = !ledBlinkState;
      digitalWrite(LED_NOTICE_PIN, ledBlinkState ? HIGH : LOW);
    }
    return;
  }

  // 2. Prioridad: Parpadeo rápido (250ms) si está intentando conectar/reconectar al Wi-Fi
  if (wifiConnecting || !isNetworkConnected()) {
    if (millis() - lastLedBlinkTime >= 250) {
      lastLedBlinkTime = millis();
      ledBlinkState = !ledBlinkState;
      digitalWrite(LED_NOTICE_PIN, ledBlinkState ? HIGH : LOW);
    }
    return;
  }

  // 3. La emergencia ilumina mientras esté vigente; los avisos conservan su enclavamiento.
  if (emergenciaEnCurso(nullptr) || ledEnclavado) {
    digitalWrite(LED_NOTICE_PIN, HIGH);
    return;
  }

  // 4. Caso normal: apagado
  digitalWrite(LED_NOTICE_PIN, LOW);
}

void handleUsbDebugInput() {
  static char buf[256];
  static uint8_t idx = 0;

  while (Serial.available() > 0) {
    char c = (char)Serial.read();
    if (c == '\n') {
      buf[idx] = 0;
      String line(buf);
      idx = 0;
      line.trim();
      if (line.length() > 0) {
        String upper = upperTrim(line);

        if (upper == "STATUS") {
          Serial.println("========== STATUS COMPLETO ==========");
          Serial.print("Red: "); Serial.println(WiFi.status() == WL_CONNECTED ? "WIFI" : "DESCONECTADO");
          Serial.print("Call: ");
          if (callWanted) Serial.println(callActive ? "ACTIVA" : "INICIANDO");
          else if (incomingRinging) Serial.println("ENTRANTE (sonando)");
          else Serial.println("IDLE");
          Serial.print("call_id: "); Serial.println(currentCallId.length() ? currentCallId : "(ninguno)");
          Serial.print("WS TX: "); Serial.println(wsTxConnected ? "CONECTADO" : "DESCONECTADO");
          Serial.print("WS RX: "); Serial.println(wsRxConnected ? "CONECTADO" : "DESCONECTADO");
          Serial.println("=====================================");
          return;
        }

        processButtonOrCmdLine(line);
      }
    } else if (c == '\r') {
      // ignore
    } else {
      if (idx < sizeof(buf) - 1) buf[idx++] = c;
      else idx = 0;
    }
  }
}

void loop() {
  handlePendingWiFiReconnect();

  if (!isNetworkConnected()) {
    wsTxCloseRequested = true;
    wsRxConnected = false;
    connectWiFi();
  }

  manejarAvisoPendiente();

  // Comprobar botones físicos
  checkButtonPin();
  checkButtonPin2();
  checkDoorButtons();

  // Control centralizado del estado del LED
  handleLedControl();

  handleUsbDebugInput();

  if (millis() - lastStats >= 1000) {
    uint32_t nowStats = millis();
    lastStats = nowStats;

    uint32_t rxB = wsRxBytes;
    uint32_t dacB = dacWrittenBytes;
    uint32_t txB = wsTxBytes;

    wsRxBytes = 0;
    dacWrittenBytes = 0;
    wsTxBytes = 0;

    Serial.print("CALL:");
    Serial.print(callWanted ? (callActive ? "ACTIVE" : "STARTING") : (incomingRinging ? "INCOMING" : "IDLE"));
    Serial.print(" | RX:"); Serial.print(rxB);
    Serial.print(" DAC:"); Serial.print(dacB);
    Serial.print(" TX:"); Serial.print(txB);
    Serial.print(" | WS:"); Serial.print(wsTxConnected ? "T" : "t");
    Serial.println(wsRxConnected ? "R" : "r");

    droppedRxBytes = 0;
    rxUnderruns = 0;
    txReadShorts = 0;
    txReadErrors = 0;
    txSendFails = 0;
  }

  delay(1);
}
